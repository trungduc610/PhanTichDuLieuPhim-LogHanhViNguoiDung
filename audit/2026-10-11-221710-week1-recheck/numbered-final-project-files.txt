FILE scripts/check_environment.py
    1: """Record actual environment and tool exit codes without installing software."""
    2: import argparse
    3: import json
    4: import platform
    5: import shutil
    6: import subprocess
    7: import sys
    8: from pathlib import Path
    9: from common import utcnow, write_json
   10: 
   11: parser = argparse.ArgumentParser(description=__doc__)
   12: parser.add_argument('--report', required=True, help='NEW environment report path')
   13: options = parser.parse_args()
   14: if Path(options.report).exists():
   15:     raise FileExistsError('Use a new report path; prior evidence will not be overwritten')
   16: result = {'checked_at_utc': utcnow(), 'os': platform.platform(), 'python': sys.version,
   17:           'python_executable': sys.executable, 'tools': {}}
   18: for args in [['java', '-version'], ['docker', 'version'], ['docker', 'compose', 'version'], ['wsl', '--status'], ['git', '--version']]:
   19:     key = ' '.join(args)
   20:     if not shutil.which(args[0]):
   21:         result['tools'][key] = {'available': False, 'reason': 'not found in PATH'}
   22:         continue
   23:     try:
   24:         proc = subprocess.run(args, capture_output=True, timeout=20)
   25:         content = proc.stdout + proc.stderr
   26:         encoding = 'utf-16-le' if b'\x00' in content else 'utf-8'
   27:         result['tools'][key] = {'available': proc.returncode == 0, 'exit_code': proc.returncode,
   28:                                'output': content.decode(encoding, errors='replace')}
   29:     except Exception as exc:
   30:         result['tools'][key] = {'available': False, 'reason': str(exc)}
   31: write_json(options.report, result)
   32: print(json.dumps(result, ensure_ascii=False, indent=2))

FILE scripts/common.py
    1: """Shared standard-library helpers; run CLI scripts from the project root."""
    2: import hashlib
    3: import json
    4: from datetime import datetime, timezone
    5: from pathlib import Path
    6: 
    7: 
    8: def sha256(path):
    9:     h = hashlib.sha256()
   10:     with open(path, 'rb') as f:
   11:         for block in iter(lambda: f.read(1024 * 1024), b''):
   12:             h.update(block)
   13:     return h.hexdigest()
   14: 
   15: 
   16: def write_json(path, obj):
   17:     path = Path(path)
   18:     path.parent.mkdir(parents=True, exist_ok=True)
   19:     path.write_text(json.dumps(obj, ensure_ascii=False, indent=2) + '\n', encoding='utf-8')
   20: 
   21: 
   22: def utcnow():
   23:     return datetime.now(timezone.utc).isoformat()
   24: 
   25: 
   26: def parse_time(value):
   27:     dt = datetime.fromisoformat(value.replace('Z', '+00:00'))
   28:     if dt.tzinfo is None:
   29:         raise ValueError('Timestamp must include timezone, e.g. +07:00 or Z')
   30:     return dt.astimezone(timezone.utc)

FILE scripts/create_delivery_manifest.py
    1: """Create or verify a scoped immutable handoff manifest (no data regeneration)."""
    2: import argparse
    3: import json
    4: from pathlib import Path
    5: from common import sha256, utcnow, write_json
    6: 
    7: ROOT=Path(__file__).resolve().parents[1]
    8: 
    9: 
   10: def verify(manifest_path):
   11:     manifest=json.loads(Path(manifest_path).read_text(encoding='utf-8'))
   12:     issues=[]
   13:     for rel, expected in manifest['files'].items():
   14:         p=ROOT/rel
   15:         if not p.is_file():
   16:             issues.append({'path':rel,'error':'missing'})
   17:         elif p.stat().st_size!=expected['bytes'] or sha256(p)!=expected['sha256']:
   18:             issues.append({'path':rel,'error':'bytes/hash mismatch'})
   19:     return {'status':'PASS' if not issues else 'FAIL','files_checked':len(manifest['files']),'issues':issues,'checked_at_utc':utcnow()}
   20: 
   21: 
   22: def main():
   23:     p=argparse.ArgumentParser(description=__doc__)
   24:     p.add_argument('--output')
   25:     p.add_argument('--verify')
   26:     p.add_argument('--audit-dir')
   27:     p.add_argument('--run',action='append',default=[])
   28:     p.add_argument('--extra-evidence',action='append',default=[])
   29:     a=p.parse_args()
   30:     if a.verify:
   31:         result=verify(a.verify)
   32:         print(json.dumps(result,ensure_ascii=False,indent=2))
   33:         return 0 if result['status']=='PASS' else 1
   34:     if not a.output or Path(a.output).exists():
   35:         raise ValueError('--output must be a NEW file')
   36:     scopes=['scripts','src','tests','config','docs','data/raw','data/processed','data/generated']
   37:     scopes += [f'evidence/{x}' for x in a.run]+[f'data/runs/{x}/output' for x in a.run]+[f'logs/{x}' for x in a.run]+a.extra_evidence
   38:     if a.audit_dir:
   39:         scopes.append(a.audit_dir)
   40:     files=set(ROOT/name for name in ['README.md','Dockerfile','compose.yaml','requirements.txt','.gitignore','.gitattributes','.dockerignore'])
   41:     output=Path(a.output).resolve()
   42:     for scope in scopes:
   43:         folder=(ROOT/scope).resolve()
   44:         if not folder.is_relative_to(ROOT):
   45:             raise ValueError('Scope outside project')
   46:         files.update(p for p in folder.rglob('*') if p.is_file())
   47:     entries={}
   48:     for file in sorted(files):
   49:         rel=file.relative_to(ROOT).as_posix()

   51:             continue

   55:                 continue  # The currently recording manifest step is not yet immutable.
   56:         entries[rel]={'bytes':file.stat().st_size,'sha256':sha256(file)}
   57:     write_json(output,{'created_at_utc':utcnow(),'root':str(ROOT),'scope':scopes,'exclusions':['.git','historical evidence/audit unless explicitly selected','channel checkpoints/data','spool copy/tracker','temporary/fixtures','pycache/part','public temporary CA bundle','active execution step','manifest itself and later verification reports'],'files':entries})
   58:     result=verify(output)
   59:     print(json.dumps(result,ensure_ascii=False,indent=2))
   60:     return 0 if result['status']=='PASS' else 1
   61: 
   62: 
   63: if __name__=='__main__':
   64:     raise SystemExit(main())

FILE scripts/download_dataset.py
    1: """Download immutable local snapshot; preserve source ZIP and license README."""
    2: import argparse
    3: import json
    4: import shutil
    5: import ssl
    6: import urllib.request
    7: import zipfile
    8: from pathlib import Path
    9: from common import sha256, utcnow, write_json
   10: 
   11: URL = 'https://files.grouplens.org/datasets/movielens/ml-latest-small.zip'
   12: 
   13: 
   14: def download(raw, ca_file=None):
   15:     raw = Path(raw)
   16:     raw.mkdir(parents=True, exist_ok=True)
   17:     archive = raw / 'ml-latest-small.zip'
   18:     manifest = raw / 'download_manifest.json'
   19:     if manifest.exists():
   20:         saved = json.loads(manifest.read_text(encoding='utf-8'))
   21:         if not archive.exists() or sha256(archive) != saved['sha256']:
   22:             raise ValueError('Existing snapshot checksum mismatch; use a NEW raw directory')
   23:         for name, digest in saved['files_sha256'].items():
   24:             if sha256(raw / name) != digest:
   25:                 raise ValueError('Raw file modified: ' + name)
   26:         print(json.dumps(saved, ensure_ascii=False, indent=2))
   27:         return saved
   28:     if archive.exists() or (raw / 'ml-latest-small').exists():
   29:         raise FileExistsError('Incomplete/untracked raw snapshot; choose a new --raw directory')
   30:     part = raw / 'download.zip.part'
   31:     if part.exists():
   32:         raise FileExistsError(part)
   33:     context = ssl.create_default_context(cafile=ca_file)
   34:     with urllib.request.urlopen(URL, context=context, timeout=120) as response, part.open('xb') as out:
   35:         shutil.copyfileobj(response, out)
   36:     with zipfile.ZipFile(part) as z:
   37:         if z.testzip() is not None:
   38:             raise ValueError('ZIP CRC check failed')
   39:         expected = {f'ml-latest-small/{x}' for x in ['README.txt', 'movies.csv', 'ratings.csv', 'tags.csv', 'links.csv']}
   40:         if not expected.issubset(z.namelist()):
   41:             raise ValueError('Unexpected ZIP contents')
   42:         # Extract ONLY explicit files: never trust archive path traversal.
   43:         for name in sorted(expected):
   44:             target = raw / name
   45:             target.parent.mkdir(parents=True, exist_ok=True)
   46:             with z.open(name) as source, target.open('xb') as out:
   47:                 shutil.copyfileobj(source, out)
   48:     part.rename(archive)
   49:     result = {'dataset': 'MovieLens latest-small', 'source_url': 'https://grouplens.org/datasets/movielens/latest/',
   50:               'download_url': URL, 'downloaded_at_utc': utcnow(), 'tls_verified': True,
   51:               'sha256': sha256(archive), 'archive_bytes': archive.stat().st_size,
   52:               'license_file': 'ml-latest-small/README.txt',
   53:               'files_sha256': {n: sha256(raw / n) for n in sorted(expected)}}
   54:     write_json(manifest, result)
   55:     print(json.dumps(result, ensure_ascii=False, indent=2))
   56:     return result
   57: 
   58: 
   59: if __name__ == '__main__':
   60:     p = argparse.ArgumentParser(description=__doc__)
   61:     p.add_argument('--raw', default='data/raw')
   62:     p.add_argument('--ca-file', help='Optional trusted PEM CA bundle; TLS verification stays enabled')
   63:     a = p.parse_args()
   64:     download(a.raw, a.ca_file)

   19: 
   20: ## Trạng thái và thời lượng
   21: 
   22: | Action | watchDuration | Ý nghĩa |
   23: |---|---:|---|
   24: | view | 0 | Mở trang/chi tiết phim, có thể rời đi mà không play |
   25: | play | 0 | Bắt đầu hoặc tiếp tục xem; không cộng lại đoạn trước |
   26: | pause | Thời gian từ play vừa trước đến pause | Đóng một đoạn xem |
   27: | stop | Thời gian từ play vừa trước đến stop | Kết thúc sớm; tổng phiên < runtime mô phỏng |
   28: | complete | Thời gian từ play vừa trước đến complete | Tổng phiên bằng runtime mô phỏng |
   29: 
   30: Chuỗi sinh: `view`; `view → play → stop/complete`; hoặc `view → play → pause → play → stop/complete`. Thời gian chờ trước play và thời gian pause không tính xem. Không có seek, tua nhanh, nhiều tab đồng thời hoặc phiên qua nhiều phim trong mô hình này. Sau khử trùng eventId, cộng watchDuration trên toàn bộ event của phiên là tổng thời gian xem.
   31: 
   32: Ví dụ diễn giải (không phải bằng chứng Flume): xem 120 giây → pause ghi 120; chờ 30 giây → play ghi 0; xem thêm 180 giây → stop ghi 180. Tổng xem 300 giây, không phải 420 hay 330 giây.
   33: 
   34: ## Phân bố và tái lập
   35: 
   36: Các phim được shuffle bằng seed rồi gán trọng số rank^-0,85. User được shuffle rồi gán trọng số rank^-0,65. Không dùng số rating để giả vờ mô phỏng phản ánh thực tế. Chọn giờ bắt đầu với trọng số 5 cho 19–23h, 0,5 cho 0–5h, 1,5 cho các giờ khác, nhân số giây hợp lệ trong khung giờ. Dự trữ bốn giờ cuối khoảng để phiên không vượt biên. Lịch sử phiên theo user dùng để loại các khoảng chồng nhau; điều này có thể làm phân bố thực tế lệch khỏi trọng số ban đầu khi dữ liệu dày.
   37: 
   38: Xấp xỉ 15% phiên chỉ view; phiên đã play có xác suất complete 55%, và xác suất có pause 35% nếu ngân sách event cho phép. Đây là tham số mô hình, không phải tỷ lệ đo từ người dùng. Nếu còn 1–2 event cuối, tạo phiên view-only để giữ đúng số lượng mà không cắt dở phiên. Một phần user/phim có thể không được chọn; `--users` là kích thước tập user, không ép tất cả phải hoạt động.
   39:

   18: 1. **Docker / Docker Compose**:

   23:    * Là tập dữ liệu metadata tĩnh mô tả thông tin phim (tên, thể loại, năm phát hành). **MovieLens không cần và không đi qua Apache Flume**, được lưu trực tiếp vào kho lưu trữ để Spark kết hợp (join) với log hành vi.
